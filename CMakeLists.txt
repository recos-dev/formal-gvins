cmake_minimum_required(VERSION 3.20)
project(formal_gvins VERSION 0.1.0 LANGUAGES CXX)

add_library(formal_gvins INTERFACE)
add_library(formal_gvins::formal_gvins ALIAS formal_gvins)
target_compile_features(formal_gvins INTERFACE cxx_std_20)
target_include_directories(formal_gvins INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")

option(FORMAL_GVINS_BUILD_EIGEN_BACKEND "Build the Eigen adapter" ON)
option(FORMAL_GVINS_BUILD_CMSIS_DSP_BACKEND "Build the adapter for externally supplied CMSIS-DSP" OFF)
set(FORMAL_GVINS_CMSIS_DSP_TARGET "CMSISDSP::CMSISDSP" CACHE STRING "Target providing CMSIS-DSP")

if(FORMAL_GVINS_BUILD_EIGEN_BACKEND)
    find_package(Eigen3 3.3 CONFIG REQUIRED)
    add_library(formal_gvins_eigen INTERFACE)
    add_library(formal_gvins::eigen ALIAS formal_gvins_eigen)
    target_link_libraries(formal_gvins_eigen INTERFACE formal_gvins Eigen3::Eigen)
endif()

if(FORMAL_GVINS_BUILD_CMSIS_DSP_BACKEND)
    if(NOT TARGET "${FORMAL_GVINS_CMSIS_DSP_TARGET}")
        find_package(CMSISDSP CONFIG REQUIRED)
    endif()
    if(NOT TARGET "${FORMAL_GVINS_CMSIS_DSP_TARGET}")
        message(FATAL_ERROR "FORMAL_GVINS_CMSIS_DSP_TARGET must name a CMSIS-DSP target")
    endif()
    add_library(formal_gvins_cmsis_dsp INTERFACE)
    add_library(formal_gvins::cmsis_dsp ALIAS formal_gvins_cmsis_dsp)
    target_link_libraries(formal_gvins_cmsis_dsp INTERFACE formal_gvins "${FORMAL_GVINS_CMSIS_DSP_TARGET}")
endif()

include(CTest)
option(FORMAL_GVINS_BUILD_FRONTEND "Build the OpenCV image frontend" OFF)
option(FORMAL_GVINS_FRONTEND_CUDA "Enable the OpenCV CUDA frontend path" OFF)
if(FORMAL_GVINS_FRONTEND_CUDA AND NOT FORMAL_GVINS_BUILD_FRONTEND)
    message(FATAL_ERROR "FORMAL_GVINS_FRONTEND_CUDA requires FORMAL_GVINS_BUILD_FRONTEND")
endif()
if(FORMAL_GVINS_BUILD_FRONTEND)
    set(FRONTEND_OPENCV_COMPONENTS core imgproc imgcodecs features2d calib3d)
    if(FORMAL_GVINS_FRONTEND_CUDA)
        list(APPEND FRONTEND_OPENCV_COMPONENTS cudafeatures2d)
    endif()
    find_package(OpenCV 4.13.0 EXACT CONFIG REQUIRED COMPONENTS ${FRONTEND_OPENCV_COMPONENTS})
    add_library(formal_gvins_frontend STATIC src/vision/orb.cpp src/vision/tracker.cpp)
    add_library(formal_gvins::frontend ALIAS formal_gvins_frontend)
    add_library(formal_gvins_euroc STATIC src/dataset/euroc.cpp)
    foreach(TARGET IN ITEMS formal_gvins_frontend formal_gvins_euroc)
        target_compile_features(${TARGET} PUBLIC cxx_std_20)
        target_include_directories(${TARGET} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
        target_include_directories(${TARGET} SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
    endforeach()
    target_link_libraries(formal_gvins_frontend PUBLIC opencv_features2d opencv_calib3d)
    target_link_libraries(formal_gvins_euroc PUBLIC opencv_imgcodecs opencv_calib3d)
    if(FORMAL_GVINS_FRONTEND_CUDA)
        target_compile_definitions(formal_gvins_frontend PUBLIC FORMAL_GVINS_FRONTEND_CUDA=1)
        target_link_libraries(formal_gvins_frontend PUBLIC opencv_cudafeatures2d)
    endif()
    add_executable(euroc_frontend examples/euroc_frontend.cpp)
    target_link_libraries(euroc_frontend PRIVATE formal_gvins_euroc formal_gvins_frontend)
    set(FEATURE_BENCHMARK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/benchmark/features")
    add_executable(orb_benchmark "${FEATURE_BENCHMARK_DIR}/orb_benchmark.cpp")
    target_link_libraries(orb_benchmark PRIVATE formal_gvins_euroc formal_gvins_frontend)
    target_compile_definitions(orb_benchmark PRIVATE
        BENCHMARK_COMPILER="${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}"
        BENCHMARK_BUILD_TYPE="$<CONFIG>")
    set(FRONTEND_TARGETS formal_gvins_frontend formal_gvins_euroc euroc_frontend orb_benchmark)
    if(BUILD_TESTING)
        find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
        add_test(NAME feature_benchmark_tools
            COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${FEATURE_BENCHMARK_DIR}/tests"
                -p "test_*.py")
        set(FORMAL_GVINS_BENCHMARK_DATASET "" CACHE PATH "EuRoC mav0 directory for the optional ORB benchmark test")
        if(FORMAL_GVINS_BENCHMARK_DATASET)
            add_test(NAME feature_benchmark_euroc
                COMMAND "${Python3_EXECUTABLE}" "${FEATURE_BENCHMARK_DIR}/run_feature_benchmark.py"
                    --executable $<TARGET_FILE:orb_benchmark> --dataset "${FORMAL_GVINS_BENCHMARK_DATASET}"
                    --output "${CMAKE_CURRENT_BINARY_DIR}/feature-benchmark-output")
            set_tests_properties(feature_benchmark_euroc PROPERTIES LABELS "dataset" TIMEOUT 660)
            if(FORMAL_GVINS_FRONTEND_CUDA)
                add_test(NAME feature_benchmark_euroc_cuda
                    COMMAND "${Python3_EXECUTABLE}" "${FEATURE_BENCHMARK_DIR}/run_feature_benchmark.py"
                        --executable $<TARGET_FILE:orb_benchmark> --dataset "${FORMAL_GVINS_BENCHMARK_DATASET}"
                        --output "${CMAKE_CURRENT_BINARY_DIR}/feature-benchmark-cuda-output" --cuda)
                set_tests_properties(feature_benchmark_euroc_cuda PROPERTIES
                    LABELS "dataset;gpu" RESOURCE_LOCK gpu TIMEOUT 660)
            endif()
        endif()
        add_executable(orb_frontend tests/test_orb.cpp)
        target_link_libraries(orb_frontend PRIVATE formal_gvins_frontend)
        add_test(NAME orb_frontend COMMAND orb_frontend)
        add_executable(euroc_reader tests/test_euroc.cpp)
        target_link_libraries(euroc_reader PRIVATE formal_gvins_euroc)
        add_test(NAME euroc_reader COMMAND euroc_reader)
        add_executable(orb_tracking tests/test_tracker.cpp)
        target_link_libraries(orb_tracking PRIVATE formal_gvins_frontend)
        add_test(NAME orb_tracking COMMAND orb_tracking)
        list(APPEND FRONTEND_TARGETS orb_frontend euroc_reader orb_tracking)
        if(FORMAL_GVINS_FRONTEND_CUDA)
            add_test(NAME orb_cuda COMMAND orb_frontend --cuda)
            add_test(NAME orb_tracking_cuda COMMAND orb_tracking --cuda)
            set_tests_properties(orb_cuda orb_tracking_cuda PROPERTIES LABELS "gpu" RESOURCE_LOCK gpu)
        endif()
    endif()
    foreach(TARGET IN LISTS FRONTEND_TARGETS)
        set_target_properties(${TARGET} PROPERTIES CXX_EXTENSIONS OFF)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endforeach()
endif()

if(BUILD_TESTING)
    option(FORMAL_GVINS_BUILD_MATH_REFERENCE_TESTS "Run independent Python mathematics checks" OFF)
    if(FORMAL_GVINS_BUILD_MATH_REFERENCE_TESTS)
        find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
        add_test(NAME msckf_equations
            COMMAND "${Python3_EXECUTABLE}" -m unittest discover
                -s "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference" -p "test_*.py")
    endif()

    function(add_numeric_test NAME SOURCE BACKEND)
        add_executable(${NAME} ${SOURCE})
        target_link_libraries(${NAME} PRIVATE ${BACKEND})
        set_target_properties(${NAME} PROPERTIES CXX_EXTENSIONS OFF)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${NAME} PRIVATE -Wall -Wextra -Wpedantic -Werror
                -fno-exceptions -fno-rtti -fno-fast-math -ffp-contract=off -UNDEBUG)
        endif()
        add_test(NAME ${NAME} COMMAND ${NAME})
    endfunction()

    add_numeric_test(scalar_standard tests/test_scalar_standard.cpp formal_gvins)
    foreach(SUITE IN ITEMS linalg linalg_spd linalg_bounded so3 so3_jacobian msckf_state)
        if(FORMAL_GVINS_BUILD_EIGEN_BACKEND)
            add_numeric_test(${SUITE}_eigen tests/test_${SUITE}.cpp formal_gvins_eigen)
            target_compile_definitions(${SUITE}_eigen PRIVATE EIGEN_RUNTIME_NO_MALLOC)
        endif()
        if(FORMAL_GVINS_BUILD_CMSIS_DSP_BACKEND)
            add_numeric_test(${SUITE}_cmsis_dsp tests/test_${SUITE}.cpp formal_gvins_cmsis_dsp)
            target_compile_definitions(${SUITE}_cmsis_dsp PRIVATE FORMAL_GVINS_TEST_CMSIS_DSP)
        endif()
    endforeach()
    if(FORMAL_GVINS_BUILD_CMSIS_DSP_BACKEND)
        add_numeric_test(cmsis_dsp_storage tests/test_cmsis_dsp_backend.cpp formal_gvins_cmsis_dsp)
    endif()
endif()
